How It Works

The Entire Interface

keyway.routes = {
    ["/users/{id}"] = {
        GET = function(ctx)
            ctx.status = 200
            ctx.headers["Content-Type"] = "application/json"
            ctx.body = '{"id":' .. ctx.params.id .. '}'
        end,
    },
}

No send(). No write(). Set fields on ctx — Lua expresses intent, Zig commits I/O.

More examples

Route Anything

["/api/users/{id}"] = {
  GET = function(ctx)
    ctx.body = get_user(ctx.params.id)
  end,
}

Middleware Per Route

["/admin/{path}"] = {
  middleware = { auth, rate_limit, log },
  GET = function(ctx)
    ctx.body = admin_panel(ctx)
  end,
}

Talk to Redis

local redis = require("keyway.redis")
local client = redis.connect()
client:set("hits", client:incr("hits"))

Query Cassandra

local cass = require("keyway.cassandra")
local db = cass.connect(config)
local rows = db:execute("SELECT * FROM messages WHERE room = ?", {room})

Call External APIs

local http = require("keyway.http_client")
local res = http.request("GET",
  "https://api.github.com/users/torvalds")
